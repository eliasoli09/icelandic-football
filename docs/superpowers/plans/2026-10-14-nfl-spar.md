import { FRANCHISE, isPlayed, type NflGame } from './types'

export interface EloParams { K: number; HFA: number; R: number; BYE: number }

/** Chosen by scripts/nfl-backtest.mts on 2002-2014; see the vault, NFL/Líkanið. */
export const NFL_ELO: EloParams = { K: 20, HFA: 48, R: 1 / 3, BYE: 25 }
export const ELO_MEAN = 1505

export const franchiseOf = (abbr: string) => FRANCHISE[abbr] ?? abbr
export const winProb = (diff: number) => 1 / (1 + 10 ** (-diff / 400))
/** 25 Elo points is worth one point on the scoreboard (FiveThirtyEight's NFL scale). */
export const eloSpread = (diff: number) => diff / 25

/** Big wins count more, less so when the favourite was expected to win big. */
export function movMultiplier(margin: number, winnerDiff: number): number {
  if (margin === 0) return 1
  return (Math.log(Math.abs(margin) + 1) * 2.2) / (winnerDiff * 0.001 + 2.2)
}

export type EloGame = Pick<NflGame, 'game_id' | 'season' | 'game_type' | 'week' | 'kickoff' | 'home' | 'away' | 'home_score' | 'away_score' | 'neutral'>

export interface EloRow { game_id: string; team: string; season: number; week: number; elo_before: number; elo_after: number }
export interface EloRecord { game_id: string; season: number; home: string; away: string; diff: number; pHome: number; margin: number }
export interface EloState { ratings: Map<string, number>; lastSeason: Map<string, number>; scheduled: Set<string>; params: EloParams }

const slot = (season: number, week: number, team: string) => `${season}:${week}:${team}`

/** Regular-season week after a week with no game for this club. */
function cameOffBye(state: EloState, g: EloGame, team: string): boolean {
  return g.game_type === 'REG' && g.week > 1 && !state.scheduled.has(slot(g.season, g.week - 1, team))
}

function rating(state: EloState, team: string, season: number, commit: boolean): number {
  let r = state.ratings.get(team) ?? ELO_MEAN
  const last = state.lastSeason.get(team)
  if (last !== undefined && last !== season) r = r * (1 - state.params.R) + ELO_MEAN * state.params.R
  if (commit) { state.ratings.set(team, r); state.lastSeason.set(team, season) }
  return r
}

function gameDiff(state: EloState, g: EloGame, eh: number, ea: number): number {
  const p = state.params
  const h = franchiseOf(g.home)
  const a = franchiseOf(g.away)
  return eh - ea + (g.neutral ? 0 : p.HFA) + (cameOffBye(state, g, h) ? p.BYE : 0) - (cameOffBye(state, g, a) ? p.BYE : 0)
}

/**
 * Fold every played game, oldest first. Unplayed games only feed the schedule
 * (bye weeks are read from it, so a future week sees the same byes as history).
 */
export function runElo(games: EloGame[], params: EloParams = NFL_ELO) {
  const state: EloState = { ratings: new Map(), lastSeason: new Map(), scheduled: new Set(), params }
  for (const g of games) if (g.game_type === 'REG') {
    state.scheduled.add(slot(g.season, g.week, franchiseOf(g.home)))
    state.scheduled.add(slot(g.season, g.week, franchiseOf(g.away)))
  }
  const played = games.filter(isPlayed).sort((a, b) => a.kickoff.localeCompare(b.kickoff) || a.game_id.localeCompare(b.game_id))
  const rows: EloRow[] = []
  const records: EloRecord[] = []
  for (const g of played) {
    const h = franchiseOf(g.home)
    const a = franchiseOf(g.away)
    const eh = rating(state, h, g.season, true)
    const ea = rating(state, a, g.season, true)
    const diff = gameDiff(state, g, eh, ea)
    const pHome = winProb(diff)
    const margin = g.home_score! - g.away_score!
    const s = margin > 0 ? 1 : margin < 0 ? 0 : 0.5
    const delta = params.K * movMultiplier(margin, margin > 0 ? diff : -diff) * (s - pHome)
    state.ratings.set(h, eh + delta)
    state.ratings.set(a, ea - delta)
    rows.push(
      { game_id: g.game_id, team: h, season: g.season, week: g.week, elo_before: eh, elo_after: eh + delta },
      { game_id: g.game_id, team: a, season: g.season, week: g.week, elo_before: ea, elo_after: ea - delta },
    )
    records.push({ game_id: g.game_id, season: g.season, home: h, away: a, diff, pHome, margin })
  }
  return { rows, records, state }
}

export interface GamePrediction { eloHome: number; eloAway: number; diff: number; pHome: number; spread: number }

/** Forecast an unplayed game from the latest ratings without changing them. */
export function predictGame(g: EloGame, state: EloState, params: EloParams = state.params): GamePrediction {
  const s = { ...state, params }
  const eloHome = rating(s, franchiseOf(g.home), g.season, false)
  const eloAway = rating(s, franchiseOf(g.away), g.season, false)
  const diff = gameDiff(s, g, eloHome, eloAway)
  return { eloHome, eloAway, diff, pHome: winProb(diff), spread: eloSpread(diff) }
}

/** Win probability implied by an American moneyline, bookmaker margin included. */
export const impliedProb = (american: number) =>
  american < 0 ? -american / (-american + 100) : 100 / (american + 100)

/** Both sides' probabilities with the margin removed proportionally. */
export function devig(homeMl: number | null, awayMl: number | null): { home: number; away: number } | null {
  if (homeMl === null || awayMl === null) return null
  const h = impliedProb(homeMl)
  const a = impliedProb(awayMl)
  return { home: h / (h + a), away: a / (h + a) }
}

import { runElo, eloSpread, type EloGame, type EloParams } from './elo'

/** Parameters are chosen on seasons up to this one and never on later ones. */
export const TUNE_LAST_SEASON = 2014

export interface Pred { p: number; s: number; spread: number; margin: number }
export interface Score { n: number; logLoss: number; brier: number; acc: number; spreadMae: number }

const EPS = 1e-12
export const ll = (p: number, s: number) => -(s * Math.log(Math.max(p, EPS)) + (1 - s) * Math.log(Math.max(1 - p, EPS)))

export function score(preds: Pred[]): Score {
  const n = preds.length
  const decided = preds.filter((x) => x.s !== 0.5 && x.p !== 0.5)
  return {
    n,
    logLoss: preds.reduce((t, x) => t + ll(x.p, x.s), 0) / n,
    brier: preds.reduce((t, x) => t + (x.p - x.s) ** 2, 0) / n,
    acc: decided.filter((x) => (x.p > 0.5) === (x.s === 1)).length / decided.length,
    spreadMae: preds.reduce((t, x) => t + Math.abs(x.spread - x.margin), 0) / n,
  }
}

export const GRID = {
  K: [15, 20, 25, 30],
  HFA: [20, 30, 40, 48, 55, 65],
  R: [0.2, 0.25, 1 / 3, 0.4, 0.5],
  BYE: [0, 15, 25, 35],
}

/** Model predictions for the played games of [from, to], from a run over everything. */
export function modelPreds(games: EloGame[], params: EloParams, from: number, to: number): (Pred & { game_id: string })[] {
  return runElo(games, params).records
    .filter((r) => r.season >= from && r.season <= to)
    .map((r) => ({ game_id: r.game_id, p: r.pHome, s: r.margin > 0 ? 1 : r.margin < 0 ? 0 : 0.5, spread: eloSpread(r.diff), margin: r.margin }))
}

/** Grid search on log loss. Throws if asked to look at a season it must not see. */
export function tune(games: EloGame[], from: number, to: number, grid = GRID): { params: EloParams; logLoss: number } {
  if (to > TUNE_LAST_SEASON) throw new Error(`Stillingar má aðeins velja á tímabilum til og með ${TUNE_LAST_SEASON}, ekki ${to}`)
  let best = { params: { K: 0, HFA: 0, R: 0, BYE: 0 }, logLoss: Infinity }
  for (const K of grid.K) for (const HFA of grid.HFA) for (const R of grid.R) for (const BYE of grid.BYE) {
    const params = { K, HFA, R, BYE }
    const l = score(modelPreds(games, params, from, to)).logLoss
    if (l < best.logLoss) best = { params, logLoss: l }
  }
  return best
}

/** Mean paired difference with a 95% interval. */
export function pairedDiff(a: number[], b: number[]): { mean: number; lo: number; hi: number } {
  const d = a.map((x, i) => x - b[i])
  const mean = d.reduce((t, x) => t + x, 0) / d.length
  const sd = Math.sqrt(d.reduce((t, x) => t + (x - mean) ** 2, 0) / (d.length - 1))
  const half = (1.96 * sd) / Math.sqrt(d.length)
  return { mean, lo: mean - half, hi: mean + half }
}

## Áfangi 2 (framkvæmdur eftir gáttina, 14. okt)

Elias valdi stillingar á 2016-2025 (K 20, HFA 40, R 0,4, BYE 35). Mælingin:
líkanið 64,4% / log loss 0,636 og Vegas 66,2% / 0,613 á 3.027 leikjum 2015-2025,
munur 0,023 [0,016, 0,031]. Val á nýlegum árum prófað á 2020-2025: −0,0007
[−0,003, +0,0015], þ.e. enginn mælanlegur munur.

Það sem var smíðað, hvert með prófum þar sem það er hreint fall:
- `006_nfl_elo.sql`: `nfl_elo` (með `p_win`, líkunum fyrir leik) og `rpc_upsert_nfl_elo`.
  `ingestNfl` endurreiknar Elo og skrifar yfirstandandi tímabil (`--all` skrifar allt).
